\begin{algorithm}[t]
\caption{\bt{Validation with shared maintenance}{带共享维护的验证}}
\label{alg:maintenance}
\small
\begin{algorithmic}[1]
\Require Revision $x$; state $D$; version stamps $V$; outcome store $\mathcal H$
\Ensure Available revision $x'$ with current evidence, or $\bot$
\Function{EvalCond}{$c, D, V, \mathcal H$}
  \IfThen{$k(c, V) \notin \mathcal H$}{$\mathcal H[k(c, V)] \gets Q_c(D)$}
  \State \Return $\operatorname{Accept}(c, \mathcal H[k(c, V)])$
\EndFunction
\Function{Validate}{$x, D, V, \mathcal H$}
  \IfThen{$x$ is invalid}{\Return $\bot$} \Comment{invalid}
  \IfThen{$V(\operatorname{dep}(x)) = V_x(\operatorname{dep}(x))$}{\Return $x$} \Comment{valid (step 3)}
  \State $\Delta C(x) \gets \{c \in C(x) \mid V(\operatorname{dep}(c)) \neq V_x(\operatorname{dep}(c))\}$
  \For{each $c \in \Delta C(x)$ (base conditions first)} \Comment{pending}
    \If{$\neg\Call{EvalCond}{c, D, V, \mathcal H}$}
      \State Invalidate $x$; \Return $\operatorname{Recover}(x, c)$
    \EndIf
  \EndFor
  \State $V_x(\operatorname{dep}(x)) \gets V(\operatorname{dep}(x))$; \Return $x$
\EndFunction
\end{algorithmic}
\end{algorithm}
