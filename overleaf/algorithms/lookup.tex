\begin{algorithm}[t]
\caption{\bt{Lookup: decide before, verify after}{查找：事前判定，事后验证}}
\label{alg:lookup}
\small
\begin{algorithmic}[1]
\Require Request $\rho$; equivalence graph $G$; snapshot $s$; tolerance $\alpha$
\Ensure An answer, a partial hit, or a miss
\Function{Lookup}{$\rho, G, s, \alpha$}
  \State $A \gets \operatorname{Parse}(\rho)$; $K \gets$ candidates for $A$ in $G$ and their neighbors
  \State $S \gets \{m \in K \mid \operatorname{Res}(\rho,m)=\emptyset \wedge m \text{ not excluded}\}$
  \If{$S = \emptyset$}
    \State \Return partial hit $(m,\operatorname{Res}(\rho,m))$ if some $m\in K$ explains the computation phrase, else miss
  \EndIf
  \State $Y \gets \{\operatorname{Run}(m,\rho,s) \mid m \in S\}$ \Comment{each validated on $s$}
  \IfThen{$\lvert Y\rvert > 1$}{\Return clarification on $\delta$ with $Y$}
  \State $g \gets$ stratum of $(\rho,S)$
  \IfThen{$U_g \le \alpha$ and no residual audit is drawn}{\Return the value in $Y$}
  \State $u \gets$ verified computation of an exploring agent \Comment{audit}
  \State \Call{Record}{$G,u,\rho$}; update $U_g$; \Return the answer of $u$
\EndFunction
\Function{Record}{$G, u, \rho$} \Comment{also after misses and at publication}
  \For{each definition $m$ on $u$'s fact table}
    \If{$I_u$ and $I_m$ meet under rules with premises $P$ holding}
      \State add equivalence edge $(u,m,P)$; add masked $\rho$ to $\Phi(m)$
    \ElsIf{$\lvert\delta(u,m)\rvert \le 2$}
      \State add difference edge $(u,m,\delta)$
    \EndIf
  \EndFor
\EndFunction
\end{algorithmic}
\end{algorithm}
