\begin{algorithm}[t]
\caption{\bt{Snapshot-bound metric execution}{快照绑定的指标执行}}
\label{alg:snapshot-execution}
\small
\begin{algorithmic}[1]
\Require Read-only SQL $q$; revisions $\mathcal R$; outcome store $\mathcal H_{\mathrm{tx}}$
\Ensure Query result $y = q(D_s)$, or $\bot$ (rejected)
\Function{ExecuteBound}{$q, \mathcal R, \mathcal H_{\mathrm{tx}}$}
  \IfThen{$\neg\operatorname{Admissible}(q, \mathcal R)$}{\Return $\bot$}
  \State $\mathcal C_{\mathcal R} \gets \bigcup_{m^r \in \mathcal R} C(m^r)$
  \State Open \textsc{repeatable read} read-only transaction $s$ on $D_s$
  \State Read transactional versions $v_s$ inside $s$
  \For{each $c \in \mathcal C_{\mathcal R}$}
    \If{$\neg\Call{EvalCond}{c, D_s, v_s, \mathcal H_{\mathrm{tx}}}$}
      \State Abort $s$; invalidate cached $V(\operatorname{dep}(c))$; \Return $\bot$
    \EndIf
  \EndFor
  \State $y \gets q(D_s)$; commit $s$; \Return $y$
\EndFunction
\end{algorithmic}
\end{algorithm}
